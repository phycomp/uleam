# 1. 建立 12 層真實規格的硬體儲存庫 (Master Repository，數據完全對接上傳檔案)
if "master_layers" not in st.session_state:
    st.session_state.master_layers = 

# 初始化預設的導體分配關係 (預設將微線路綁定在真實規格的「訊號層」)
if "conductors_data" not in st.session_state:
    st.session_state.conductors_data = [
        {"target_layer": "訊號層", "mode": "absolute", "name": "RF_Trace_P", "x_center_or_start": -80.0, "width": 50.0, "spacing": 0.0, "count": 1},
        {"target_layer": "訊號層", "mode": "absolute", "name": "RF_Trace_N", "x_center_or_start": 20.0, "width": 50.0, "spacing": 0.0, "count": 1},
    ]

# 初始化盲埋孔集合 (支援縱向跨越，例如從訊號層拉通孔到上通孔鍍層)
if "vias_data" not in st.session_state:
    st.session_state.vias_data = [
        {"name": "Laser_Via_Top", "type": "blind", "start_layer": "上通孔鍍層-銅", "end_layer": "訊號層", "x_center": -80.0, "drill_diameter": 30.0, "color": "#e64a19"}
    ]

# 2. 側邊欄允許動態微調激活層數 (上限滑桿自動適配擴充至 12 層)
num_target_layers = st.sidebar.slider("🎚️ 設定材質疊構層數 (Max 12)", min_value=1, max_value=12, value=12)

# 依據輸入層數，動態擷取對應數量的真實材質層
active_stackup = st.session_state.master_layers[:num_target_layers]

# ==============================================================================
# 1. 介面模式切換：雙軌並行模式開關
# ==============================================================================
editor_mode = st.radio("🛠️ 選擇操作模式 (Editor Mode)", ["UI 表單配置驅動", "進階純文字手寫手動調整 (解鎖原始 AST)"], horizontal=True)

# 依據新導入的 12 層中文命名，動態更新跨表格選單清單
valid_layer_names = [ly["layer_name"] for ly in active_stackup]
conductor_layer_names = [ly["layer_name"] for ly in active_stackup if ly["type"] == "conductor_layer"]

# ==============================================================================
# 2. 模式 A：UI 表單配置驅動
# ==============================================================================
if editor_mode == "UI 表單配置驅動":
    col_ui, col_ast = st.columns()
    
    with col_ui:
        st.subheader(f"📐 (1) 疊構材質配置 (Stackup Registry) - 當前已啟用 {num_target_layers} 層")
        with st.expander(f"展開檢視/調整當前 {num_target_layers} 層基本參數 (已導入真實 PI/NbTi 規格)", expanded=True):
            for idx in range(num_target_layers):
                ly = active_stackup[idx]
                c1, c2, c3, c4 = st.columns()
                with c1:
                    ly["layer_name"] = st.text_input(f"Layer {idx+1} 名稱", value=ly["layer_name"], key=f"ly_name_{idx}")
                with c2:
                    ly["type"] = st.selectbox(f"類型 #{idx+1}", ["dielectric", "conductor_layer"], index=0 if ly["type"]=="dielectric" else 1, key=f"ly_type_{idx}")
                with c3:
                    # 原本的整數限制升級為浮點數以精確適配上介電膠的 12.5um 或金屬層的 0.2um
                    ly["thickness"] = st.number_input(f"厚度 (μm) #{idx+1}", min_value=0.1, max_value=1000.0, value=float(ly["thickness"]), step=0.5, key=f"ly_thick_{idx}")
                if ly["type"] == "conductor_layer":
                    with c4:
                        ly["etch_factor"] = st.slider(f"蝕刻因子 (deg) #{idx+1}", 45, 90, int(ly["etch_factor"]), key=f"ly_etch_{idx}")
                else:
                    ly["etch_factor"] = 90
